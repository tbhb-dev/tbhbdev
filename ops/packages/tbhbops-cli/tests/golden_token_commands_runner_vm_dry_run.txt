set -eu

# Dry run: each op command below is echoed, not run.
# It first creates each vault it needs that the account lacks.
# It grants the Connect server srv the vault runner-fixture and each lab node's vault,
# then creates one token with no expiry for each CI runner VM.
# The token reads only the vault runner-fixture, read-only. It is kept
# in its lab node's vault as the item runner-access-token, field token.
# A VM whose node vault already holds it is skipped, so the script is
# safe to run again.

# Live, it lists the vaults once and stops with exit 1 if that fails:
echo 'op vault list --account acct.example.com --format json'
# Live, it creates each of these vaults that the list lacks, and prints
# created vault <name> for each one:
echo 'op vault create --account acct.example.com runner-fixture'
echo 'op vault create --account acct.example.com host-ex-lab1'

echo 'op connect vault grant --account acct.example.com --server srv --vault runner-fixture'
echo 'op connect vault grant --account acct.example.com --server srv --vault host-ex-lab1'

# runner-vm ex-lab1-ci
# Live, it skips this host if this succeeds:
echo 'op item get --account acct.example.com runner-access-token --vault host-ex-lab1'
echo 'tok=$(op connect token create --account acct.example.com '"'"'runner-vm:ex-lab1-ci:20261108T142100Z'"'"' --server srv --vault runner-fixture,r)'
# Live, it stops with exit 1 if the token is empty:
echo 'printf '"'"'%s'"'"' "$tok" | jq -R '"'"'{title: "runner-access-token", category: "API_CREDENTIAL", fields: [{label: "token", type: "CONCEALED", value: .}]}'"'"' | op item create --account acct.example.com --vault host-ex-lab1 -'
echo 'unset tok'
