set -eu

# Dry run: each op and ssh command below is echoed, not run.
# It first creates each vault it needs that the account lacks.
# It grants the Connect server srv its vaults, then creates one token with no expiry for each host
# and stores it in the host's vault as the item connect-token, field token.
# A host whose vault already holds connect-token is skipped, so the script is safe
# to run again.
# Each operator Mac also gets an ops-controller token, read-only on the
# vaults of the ops accounts. It goes into the ops user's
# keychain over ssh as that user, and into no vault. A Mac where that
# user already holds it is skipped.

# Live, it lists the vaults once and stops with exit 1 if that fails:
echo 'op vault list --account acct.example.com --format json'
# Live, it creates each of these vaults that the list lacks, and prints
# created vault <name> for each one:
echo 'op vault create --account acct.example.com host-0-shared'
echo 'op vault create --account acct.example.com host-ex-ctl1'
echo 'op vault create --account acct.example.com host-ex-ctl2'
echo 'op vault create --account acct.example.com host-ex-lab1'
echo 'op vault create --account acct.example.com host-ex-win1'
echo 'op vault create --account acct.example.com host-storage-nas-a'
echo 'op vault create --account acct.example.com host-storage-nas-b'

echo 'op connect vault grant --account acct.example.com --server srv --vault host-0-shared'
echo 'op connect vault grant --account acct.example.com --server srv --vault host-ex-ctl1'
echo 'op connect vault grant --account acct.example.com --server srv --vault host-ex-ctl2'
echo 'op connect vault grant --account acct.example.com --server srv --vault host-ex-lab1'
echo 'op connect vault grant --account acct.example.com --server srv --vault host-ex-win1'
echo 'op connect vault grant --account acct.example.com --server srv --vault host-storage-nas-a'
echo 'op connect vault grant --account acct.example.com --server srv --vault host-storage-nas-b'

# ex-ctl1
# Live, it skips this host if this succeeds:
echo 'op item get --account acct.example.com connect-token --vault host-ex-ctl1'
echo 'tok=$(op connect token create --account acct.example.com '"'"'agent-host:ex-ctl1:20261108T142100Z'"'"' --server srv --vault host-0-shared --vault host-ex-ctl1,r --vault host-ex-ctl2,r --vault host-ex-lab1,r --vault host-ex-win1,r)'
# Live, it stops with exit 1 if the token is empty:
echo 'printf '"'"'%s'"'"' "$tok" | jq -R '"'"'{title: "connect-token", category: "API_CREDENTIAL", fields: [{label: "token", type: "CONCEALED", value: .}]}'"'"' | op item create --account acct.example.com --vault host-ex-ctl1 -'
echo 'unset tok'

# ex-ctl2
# Live, it skips this host if this succeeds:
echo 'op item get --account acct.example.com connect-token --vault host-ex-ctl2'
echo 'tok=$(op connect token create --account acct.example.com '"'"'agent-host:ex-ctl2:20261108T142100Z'"'"' --server srv --vault host-0-shared --vault host-ex-ctl1,r --vault host-ex-ctl2,r --vault host-ex-lab1,r --vault host-ex-win1,r)'
# Live, it stops with exit 1 if the token is empty:
echo 'printf '"'"'%s'"'"' "$tok" | jq -R '"'"'{title: "connect-token", category: "API_CREDENTIAL", fields: [{label: "token", type: "CONCEALED", value: .}]}'"'"' | op item create --account acct.example.com --vault host-ex-ctl2 -'
echo 'unset tok'

# ex-lab1
# Live, it skips this host if this succeeds:
echo 'op item get --account acct.example.com connect-token --vault host-ex-lab1'
echo 'tok=$(op connect token create --account acct.example.com '"'"'agent-host:ex-lab1:20261108T142100Z'"'"' --server srv --vault host-ex-lab1,r --vault host-0-shared,r)'
# Live, it stops with exit 1 if the token is empty:
echo 'printf '"'"'%s'"'"' "$tok" | jq -R '"'"'{title: "connect-token", category: "API_CREDENTIAL", fields: [{label: "token", type: "CONCEALED", value: .}]}'"'"' | op item create --account acct.example.com --vault host-ex-lab1 -'
echo 'unset tok'

# ex-win1
# Live, it skips this host if this succeeds:
echo 'op item get --account acct.example.com connect-token --vault host-ex-win1'
echo 'tok=$(op connect token create --account acct.example.com '"'"'agent-host:ex-win1:20261108T142100Z'"'"' --server srv --vault host-ex-win1,r --vault host-0-shared,r)'
# Live, it stops with exit 1 if the token is empty:
echo 'printf '"'"'%s'"'"' "$tok" | jq -R '"'"'{title: "connect-token", category: "API_CREDENTIAL", fields: [{label: "token", type: "CONCEALED", value: .}]}'"'"' | op item create --account acct.example.com --vault host-ex-win1 -'
echo 'unset tok'

# ops-controller ex-ctl1
# Live, it prepares the ops keychain ops-connect.keychain on ex-ctl1:
echo 'echo '"'"'mkdir -p ~/Library/Keychains && { [ -e ~/Library/Keychains/ops-connect.keychain-db ] || security create-keychain -p "" ops-connect.keychain; } && security unlock-keychain -p "" ops-connect.keychain && security set-keychain-settings ops-connect.keychain'"'"' | ssh -o IdentityAgent=none -o IdentitiesOnly=yes -i "$HOME/.ssh/sessions/$(hostname -s)_ed25519" ops@ex-ctl1 '"'"'sh -s'"'"''
# Live, it skips this Mac if ops holds ops-connect:
echo 'ssh -n -o IdentityAgent=none -o IdentitiesOnly=yes -i "$HOME/.ssh/sessions/$(hostname -s)_ed25519" ops@ex-ctl1 '"'"'security find-generic-password -s ops-connect -a agent ops-connect.keychain'"'"''
echo 'tok=$(op connect token create --account acct.example.com '"'"'ops-controller:ex-ctl1:20261108T142100Z'"'"' --server srv --vault host-storage-nas-a,r --vault host-storage-nas-b,r)'
# Live, it stops with exit 1 if the token is empty, odd or too long:
# Live, it stores the token in one ssh session that unlocks the keychain:
echo 'echo "add-generic-password -U -s ops-connect -a agent -T /usr/bin/security -w $tok ops-connect.keychain" | ssh -o IdentityAgent=none -o IdentitiesOnly=yes -i "$HOME/.ssh/sessions/$(hostname -s)_ed25519" ops@ex-ctl1 '"'"'security unlock-keychain -p "" ops-connect.keychain && security -i'"'"''
# Live, if the token is over-length or not stored, it deletes the token
# and stops with exit 1:
echo 'op connect token delete --account acct.example.com '"'"'ops-controller:ex-ctl1:20261108T142100Z'"'"' --server srv'
echo 'unset tok'
# Live, it stops with exit 1 unless the item is there:
echo 'ssh -n -o IdentityAgent=none -o IdentitiesOnly=yes -i "$HOME/.ssh/sessions/$(hostname -s)_ed25519" ops@ex-ctl1 '"'"'security find-generic-password -s ops-connect -a agent ops-connect.keychain'"'"''

# ops-controller ex-ctl2
# Live, it prepares the ops keychain ops-connect.keychain on ex-ctl2:
echo 'echo '"'"'mkdir -p ~/Library/Keychains && { [ -e ~/Library/Keychains/ops-connect.keychain-db ] || security create-keychain -p "" ops-connect.keychain; } && security unlock-keychain -p "" ops-connect.keychain && security set-keychain-settings ops-connect.keychain'"'"' | ssh -o IdentityAgent=none -o IdentitiesOnly=yes -i "$HOME/.ssh/sessions/$(hostname -s)_ed25519" ops@ex-ctl2 '"'"'sh -s'"'"''
# Live, it skips this Mac if ops holds ops-connect:

echo 'tok=$(op connect token create --account acct.example.com '"'"'ops-controller:ex-ctl2:20261108T142100Z'"'"' --server srv --vault host-storage-nas-a,r --vault host-storage-nas-b,r)'
# Live, it stops with exit 1 if the token is empty, odd or too long:

# Live, it stops with exit 1 unless the item is there:
